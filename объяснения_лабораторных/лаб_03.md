  8 | def build_clean_orders(df: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
  9 |     """Возвращает (clean_orders, quarantine_orders)."""
 10 |     out = df.copy()
 11 |     for col in DATE_COLS:
 12 |         out[col] = pd.to_datetime(out[col], errors="coerce")
 13 | 
 14 |     delivered = out["order_status"] == "delivered"
 15 |     delivery_date = out["order_delivered_customer_date"]
 16 |     purchase_date = out["order_purchase_timestamp"]
 17 |     reason = pd.Series(pd.NA, index=out.index, dtype="object")
 18 |     reason = reason.mask(delivered & delivery_date.isna(), "R1")
 19 |     reason = reason.mask(~delivered & delivery_date.notna(), "R2")
 20 |     reason = reason.mask(delivery_date < purchase_date, "R3")
 21 |     reason = reason.mask(out["order_approved_at"] < purchase_date, "R4")
 22 | 
 23 |     quarantine = out[reason.notna()].copy()
 24 |     quarantine["reason"] = reason[reason.notna()]
 25 |     clean = out[reason.isna()].copy()
 26 |     clean["is_delivered"] = clean["order_status"] == "delivered"
 27 |     clean["delivery_days"] = (
 28 |         clean["order_delivered_customer_date"] - clean["order_purchase_timestamp"]
 29 |     ).dt.days.where(clean["is_delivered"]).astype("Int64")
 30 |     clean["is_late"] = (
 31 |         clean["order_delivered_customer_date"] > clean["order_estimated_delivery_date"]
 32 |     ).astype("boolean").where(clean["is_delivered"])
 33 |     return clean.reset_index(drop=True), quarantine.reset_index(drop=True)
