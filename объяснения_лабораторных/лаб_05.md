  1 | def t_build_mart(ctx: dict) -> int:

  6 |     con.execute("""

 15 |

 18 |         con.execute("DELETE FROM mart.daily_sales WHERE order_date BETWEEN ? AND ?", [start, end])

 20 |             INSERT INTO mart.daily_sales
 21 |             SELECT order_date,
 22 |                    COUNT(DISTINCT order_id) AS n_orders,
